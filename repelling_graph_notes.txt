void move_nbrs (NodeMap nodes, size_t node_id) {
	num_nbrs = nodes[node_id].nbrs.length();
	for (auto nbr_id : nodes[node_id].nbrs) {
		nodes[nbr_id].pos -= distance (nodes[nbr_id].pos, nodes[(nbr_id + 1) % num_nbrs].pos);
		nodes[nbr_id].pos += distance (nodes[nbr_id].pos, nodes[(nbr_id - 1) % num_nbrs].pos);
		nodes[nbr_id].pos = EDGE_LEN * normalise ( distance ( nodes[nbr_id].pos, nodes[node_id].pos ) );
		move_nbrs(nbr_id);
	}
}
